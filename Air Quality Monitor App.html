  <script>

    // REPLACE WITH YOUR PUBLISHED CSV URL

    const SHEET_CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vR0_D16yF2BQM3G8xVAn4ZFjIug0xzsyVx5P3O408GsAIwvZBTTy6K478pg_K5r603o5GJZ0mHnK0Q3/pub?output=csv'; 


    let chart;


    function initChart() {

      const ctx = document.getElementById('trendChart').getContext('2d');

      chart = new Chart(ctx, {

        type: 'line',

        data: {

          labels: [],

          datasets: [{

            label: 'CO2 (ppm)',

            data: [],

            borderColor: '#3b82f6',

            borderWidth: 2,

            tension: 0.3,

            pointRadius: 0

          }]

        },

        options: {

          responsive: true,

          maintainAspectRatio: false,

          scales: {

            x: { display: false },

            y: { grid: { color: '#374151' }, ticks: { color: '#9ca3af', font: { size: 10 } } }

          },

          plugins: { legend: { display: false } }

        }

      });

    }


    function fetchSheetData() {

      Papa.parse(SHEET_CSV_URL, {

        download: true,

        header: true,

        complete: function(results) {

          const data = results.data.filter(row => row.Timestamp);

          if (data.length === 0) return;


          // Get the absolute latest record (Last row)

          const latest = data[data.length - 1];


          // Map CSV Headers (assuming standard form order)

          const keys = Object.keys(latest);

          document.getElementById('val-co2').innerText = latest[keys[1]] || '0';

          document.getElementById('val-temp').innerText = parseFloat(latest[keys[2]] || 0).toFixed(1);

          document.getElementById('val-rh').innerText = parseFloat(latest[keys[3]] || 0).toFixed(0);

          document.getElementById('val-pm1').innerText = latest[keys[4]] || '0';

          document.getElementById('val-pm25').innerText = latest[keys[5]] || '0';

          document.getElementById('val-pm25-sub').innerText = latest[keys[5]] || '0';

          document.getElementById('val-pm4').innerText = latest[keys[6]] || '0';

          document.getElementById('val-pm10').innerText = latest[keys[7]] || '0';

          

          document.getElementById('last-updated').innerText = 'Updated: ' + latest[keys[0]];


          // Update Trend Chart (Last 30 rows)

          const history = data.slice(-30);

          chart.data.labels = history.map(r => r[keys[0]]);

          chart.data.datasets[0].data = history.map(r => parseFloat(r[keys[1]]) || 0);

          chart.update();

        }

      });

    }


    // Initialize & Set 30-Second Auto Refresh

    window.onload = () => {

      initChart();

      fetchSheetData();

      setInterval(fetchSheetData, 30000);

    };

  </script>
